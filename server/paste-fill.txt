bash <<'OKX_FILL_INSTALL'
#!/bin/bash
# One-shot installer (Debian, root): every 2 min, trigger the okx-fill-watch GitHub Actions workflow.
# Reuses the token already saved by install.sh in /etc/okx-dispatch/token (no new token needed).
# Idempotent: safe to re-run. Does not touch the existing okx-dispatch (price watch) timer.
set -euo pipefail
REPO="boostemotion/okx-price-watch"
WORKFLOW="fill-watch.yml"
REF="main"

command -v curl >/dev/null || { apt-get update -qq && apt-get install -y -qq curl; }
[ -s /etc/okx-dispatch/token ] || { echo "ERROR: /etc/okx-dispatch/token missing - run the price-watch paste (paste.txt) first"; exit 1; }

cat > /usr/local/bin/okx-fill-dispatch <<SCRIPT
#!/bin/bash
# Past 2026-11-23 00:00 UTC+8 the watch window is over: turn this timer off.
if [ "\$(date +%s)" -ge 1795363200 ]; then echo "window over; disabling timer"; systemctl disable --now okx-fill-dispatch.timer; exit 0; fi
# POST workflow_dispatch; GitHub answers 204 on success. Output goes to the journal.
code=\$(curl -sS -o /tmp/okx-fill-dispatch.out -w '%{http_code}' -m 20 -X POST \\
  -H "Authorization: Bearer \$(cat /etc/okx-dispatch/token)" \\
  -H "Accept: application/vnd.github+json" -H "X-GitHub-Api-Version: 2022-11-28" \\
  https://api.github.com/repos/$REPO/actions/workflows/$WORKFLOW/dispatches \\
  -d '{"ref":"$REF"}') || code=000
if [ "\$code" = 204 ]; then echo "dispatch ok"; else echo "dispatch FAILED http=\$code \$(head -c 300 /tmp/okx-fill-dispatch.out 2>/dev/null)"; exit 1; fi
SCRIPT
chmod 755 /usr/local/bin/okx-fill-dispatch

cat > /etc/systemd/system/okx-fill-dispatch.service <<'UNIT'
[Unit]
Description=Trigger okx-fill-watch GitHub Actions workflow
Wants=network-online.target
After=network-online.target
[Service]
Type=oneshot
ExecStart=/usr/local/bin/okx-fill-dispatch
UNIT

cat > /etc/systemd/system/okx-fill-dispatch.timer <<'UNIT'
[Unit]
Description=okx-fill-watch dispatch every 2 min (stops 2026-11-23)
[Timer]
OnCalendar=*:0/2
Persistent=true
AccuracySec=5s
[Install]
WantedBy=timers.target
UNIT

systemctl daemon-reload
systemctl enable --now okx-fill-dispatch.timer
systemctl start okx-fill-dispatch.service || true
journalctl -u okx-fill-dispatch.service -n 3 --no-pager
systemctl list-timers okx-fill-dispatch.timer --no-pager
OKX_FILL_INSTALL
